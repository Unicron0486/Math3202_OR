	\section*{NEO Detections}
	
	\subsection*{Sets}
	
	\begin{tabular}{rl}
		$T$ & Telescopes\\
		$B$ & Belts\\
		$D$ & Days
	\end{tabular}

	\subsection*{Data}
	
	\begin{tabular}{rl}
		$e_t$ & Efficiency of Neo detections of telescope $t\in T$\\
		$n_d$ & Hours Neo belt is visible on day $d\in D$\\
		$h_d$ & Total hours available to observe the main or neo belt on day $d\in D$\\
		$r_b$ & The detection rate of NEOs in belt $b\in B$\\
		$p$ & processing rate: 25Gb/hr of footage\\
		$c$ & processing capacity: 500Gb\\
	\end{tabular}

	\subsection*{Variables}
	
	\begin{tabular}{rl}
		$x_{dt}$ & How many hours a day $d\in D$ each telescope $t\in T$ observes the the NEO belt\\
		$y_{dt}$ & How many hours a day $d\in D$ each telescope $t\in T$ observes the the main belt\\
		$processed_{d}$ & The amount of data processed on day $d\in D$\\
		$stored_{d}$ & The amount of raw data stored on day $d\in D$
	\end{tabular}

	\subsection*{Objective}
	
	\begin{equation*}
	\max \sum_{d\in D}\sum_{t\in T}E_t(r_{neo} x_{dt} + r_{main} y_{dt})
	\end{equation*}
	
	\subsection*{Constraints}
	
	\begin{align}
		x_{dt} \leq N_d && \forall t\in T, \forall d\in D\label{C1}\\
		% comm 2
		y_{dt} \leq H_d - x_{dt} && \forall t\in \{T | t \notin (SA1, CL2)\}, \forall d\in D\label{C2}\\
		y_{dt} \leq H_d && \forall t\in \{T | t \in (SA1, CL2)\}, \forall d\in D\label{C3}\\
		y_{dt} \geq N_d && \forall t\in \{T | t \in (SA1, CL2)\}, \forall d\in D\label{C4}\\
		% comm 3
		% p(x_{dt} + y_{dt}) \leq c && \forall t\in \{T | t \notin (SA1, CL2)\}, \forall d\in D\label{C5}\\
		% p(y_{dt}) \leq c && \forall t\in \{T | t \in (SA1, CL2)\}, \forall d\in D\label{C6}\\
		% comm 4
		\sum_{d\in D} x_{dt} + y_{dt} \geq 10 && \forall t \in T\label{C7}\\
		(processed_{d} + stored_{d-1}) . p \leq c && \forall t\in \{T | t \notin (SA1, CL2)\}, \forall d\in D\label{C5}\\
		processed_{d} . p \leq c && \forall t\in \{T | t \in (SA1, CL2)\}, \forall d\in D\label{C6}\\
		% comm5
		observation_{d} = \sum_{t\in T} x_{dt} + y_{dt} &&\forall t\in \{T | t \notin (SA1, CL2)\}, \forall d\in D\label{C15}\\
		observation_{d} = \sum_{t\in T} y_{dt} &&\forall t\in \{T | t \in (SA1, CL2)\}, \forall d\in D\label{C14}\\
		stored_{d} = observation_{d} - processed_{d} && \forall d\in D\label{C8}\\
		stored_{d} . p \leq c && \forall d\in D\label{C9}\\
		stored_{d} \geq 0 && \forall d\in D\label{C12}\\
		stored_{d} \leq observation_{d} && \forall d\in D\label{C13}\\
		%non neg
		x_{dt} \geq 0 && \forall t\in T, \forall d\in D\label{C10}\\
		y_{dt} \geq 0 && \forall t\in T, \forall d\in D\label{C11}
	\end{align}
	
	
	% Constraints (\ref{C1}) say that the amount mined from a deposit must not exceed its capacity, and constraints (\ref{C2}) ensure that each customer receives the tonnage they have requested. Constraints (\ref{C3}-\ref{C4}) enforce the minimum aluminium and silicon requirements respectively.
	Constraints (\ref{C1}) states the Processed NEO data must be no more than the available time to observe the NEO belt.
	Similarly (\ref{C2}) restricts the main belt processing to be the remaining observation time once the NEO belt has been observed. This is applied to all telescopes except SA1 and CL2 as they observe both belts simultaneously.
	To address the simultaneous observation capabilities of SA1 and CL2 (\ref{C3}-\ref{C4}) make the total main belt observation equal to the NEO belt observation with the additional remaining time to observe the main belt.
	(\ref{C5}-\ref{C6}) restrict the daily processing limit with SA1 and CL2 able to process the NEO and main belt footage together as it is one image.
	The sumation in (\ref{C7}) specifys that all telescopes must be used at least 10 hours a week.
	The constraints set out in (\ref{C8}-\ref{C13}) are specifically to address the use of additional storage and its maximum capacity. 
	(\ref{C14}-\ref{C15}) simply define how $observation_{d}$ is calculated for the use of storage as the telescopes that can observe 
	both belts simultaneously should not count observation time of both belts as it is one observation on both belts.
	(\ref{C8}) is crucial as it defines where this storage can come from, the unporcessed observation footage.
	Finally the non negative restrictions are applied to both variables in (\ref{C10}-\ref{C11})
